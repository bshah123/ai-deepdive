# RLHF (Reinforcement Learning from Human Feedback) with PPO

The classical RLHF pipeline (Christiano et al., 2017; InstructGPT, 2022) trains a policy $\pi_\theta$ using **Proximal Policy Optimization (PPO)** against a learned **Reward Model $r_\psi(x, y)$**:

$$\max_{\theta} \mathbb{E}_{(x, y) \sim \mathcal{D}_{\pi_\theta}} \left[ r_\psi(x, y) \right] - \beta \, \mathbb{D}_{\text{KL}}\left( \pi_\theta(y \mid x) \,\|\, \pi_{\text{ref}}(y \mid x) \right)$$

### The Complexity Bottleneck of PPO:
Running PPO requires holding **4 full LLM instances in GPU memory simultaneously** (Actor, Critic, Reward, and Reference model), leading to extreme VRAM demands and notorious training instability (mode collapse, reward hacking).

---

# Direct Preference Optimization (DPO)

**Direct Preference Optimization (DPO)** (Rafailov et al., Stanford 2023) mathematically proves that the optimal policy under the Bradley-Terry preference model can be expressed analytically, **completely eliminating the need for a separate Reward Model and PPO reinforcement learning loop**!

### The Closed-Form Implicit Reward Derivation
Under the KL-regularized RL objective, the theoretical optimal policy $\pi^*$ satisfies:

$$\pi^*(y \mid x) = \frac{1}{Z(x)} \pi_{\text{ref}}(y \mid x) \exp\left(\frac{1}{\beta} r(x, y)\right)$$

Rearranging for the ground-truth reward $r(x, y)$:

$$r(x, y) = \beta \log \frac{\pi^*(y \mid x)}{\pi_{\text{ref}}(y \mid x)} + \beta \log Z(x)$$

Substituting this implicit reward into the Bradley-Terry preference probability $P(y_w \succ y_l \mid x) = \sigma(r(x, y_w) - r(x, y_l))$ cancels the unknown partition function $Z(x)$, yielding the **DPO Loss Function**:

$$\mathcal{L}_{\text{DPO}}(\theta; \pi_{\text{ref}}) = -\mathbb{E}_{(x, y_w, y_l) \sim \mathcal{D}} \left[ \log \sigma \left( \beta \log \frac{\pi_\theta(y_w \mid x)}{\pi_{\text{ref}}(y_w \mid x)} - \beta \log \frac{\pi_\theta(y_l \mid x)}{\pi_{\text{ref}}(y_l \mid x)} \right) \right]$$

# Comparison of Alignment Algorithms

| Algorithm | Requires Reward Model? | Requires Critic/Value Model? | GPU VRAM Overhead | Training Stability | Primary Use Cases |
|---|---|---|---|---|---|
| **RLHF (PPO)** | Yes (Trained first) | Yes | **4x Model Parameters** | Moderate (Hyperparameter sensitive) | OpenAI GPT-4, Anthropic Claude |
| **DPO** | **No** (Direct implicit reward) | **No** | **2x Model Parameters** (Policy + Ref) | **Extremely High** (Stable classification) | **LLaMA-3, Mistral, Zephyr** |
| **KTO** | No | No | 2x Model Parameters | High | Datasets with binary Thumbs-Up/Down labels |
| **ORPO** | No | **No** (Zero Ref Model needed!) | **1x Model Parameter** | High | Ultra-low VRAM fine-tuning |

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, Dataset
from typing import List, Dict, Optional, Tuple
import time
import numpy as np

class SFTDataset(Dataset):
    """Dataset for Supervised Fine-Tuning (SFT)."""
    def __init__(self, inputs: List[torch.Tensor], labels: List[torch.Tensor]):
        self.inputs = inputs
        self.labels = labels

    def __len__(self) -> int:
        return len(self.inputs)

    def __getitem__(self, idx: int) -> Tuple[torch.Tensor, torch.Tensor]:
        return self.inputs[idx], self.labels[idx]

class RLHFModel(nn.Module):
    """Model for Reinforcement Learning from Human Feedback (RLHF)."""
    def __init__(self, input_dim: int, hidden_dim: int, output_dim: int):
        super(RLHFModel, self).__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
        self.relu = nn.ReLU()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        return x

class DPOModel(nn.Module):
    """Model for Direct Preference Optimization (DPO)."""
    def __init__(self, input_dim: int, hidden_dim: int, output_dim: int):
        super(DPOModel, self).__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
        self.relu = nn.ReLU()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        return x

def train_sft(model: nn.Module, dataloader: DataLoader, criterion: nn.Module, optimizer: optim.Optimizer, epochs: int) -> None:
    """Train the model using Supervised Fine-Tuning (SFT)."""
    model.train()
    for epoch in range(epochs):
        for inputs, labels in dataloader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
        print(f'Epoch {epoch+1}, Loss: {loss.item()}')

def train_rlhf(model: nn.Module, dataloader: DataLoader, criterion: nn.Module, optimizer: optim.Optimizer, epochs: int) -> None:
    """Train the model using Reinforcement Learning from Human Feedback (RLHF)."""
    model.train()
    for epoch in range(epochs):
        for inputs, labels in dataloader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
        print(f'Epoch {epoch+1}, Loss: {loss.item()}')

def train_dpo(model: nn.Module, dataloader: DataLoader, criterion: nn.Module, optimizer: optim.Optimizer, epochs: int) -> None:
    """Train the model using Direct Preference Optimization (DPO)."""
    model.train()
    for epoch in range(epochs):
        for inputs, labels in dataloader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
        print(f'Epoch {epoch+1}, Loss: {loss.item()}')

if __name__ == "__main__":
    # Generate synthetic data
    input_dim = 10
    hidden_dim = 20
    output_dim = 5
    num_samples = 1000
    inputs = [torch.randn(input_dim) for _ in range(num_samples)]
    labels = [torch.randint(0, output_dim, (1,)).float() for _ in range(num_samples)]

    # Create datasets and dataloaders
    sft_dataset = SFTDataset(inputs, labels)
    sft_dataloader = DataLoader(sft_dataset, batch_size=32, shuffle=True)

    rlhf_dataset = SFTDataset(inputs, labels)
    rlhf_dataloader = DataLoader(rlhf_dataset, batch_size=32, shuffle=True)

    dpo_dataset = SFTDataset(inputs, labels)
    dpo_dataloader = DataLoader(dpo_dataset, batch_size=32, shuffle=True)

    # Initialize models, criteria, and optimizers
    sft_model = RLHFModel(input_dim, hidden_dim, output_dim)
    rlhf_model = RLHFModel(input_dim, hidden_dim, output_dim)
    dpo_model = DPOModel(input_dim, hidden_dim, output_dim)

    criterion = nn.CrossEntropyLoss()
    sft_optimizer = optim.Adam(sft_model.parameters(), lr=0.001)
    rlhf_optimizer = optim.Adam(rlhf_model.parameters(), lr=0.001)
    dpo_optimizer = optim.Adam(dpo_model.parameters(), lr=0.001)

    # Train models
    print("Training SFT Model...")
    start_time = time.time()
    train_sft(sft_model, sft_dataloader, criterion, sft_optimizer, epochs=10)
    end_time = time.time()
    print(f"Training completed in {end_time - start_time:.2f} seconds")

    print("\nTraining RLHF Model...")
    start_time = time.time()
    train_rlhf(rlhf_model, rlhf_dataloader, criterion, rlhf_optimizer, epochs=10)
    end_time = time.time()
    print(f"Training completed in {end_time - start_time:.2f} seconds")

    print("\nTraining DPO Model...")
    start_time = time.time()
    train_dpo(dpo_model, dpo_dataloader, criterion, dpo_optimizer, epochs=10)
    end_time = time.time()
    print(f"Training completed in {end_time - start_time:.2f} seconds")
