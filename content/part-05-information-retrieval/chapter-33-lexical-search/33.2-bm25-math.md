import math
from typing import List, Dict, Tuple
from collections import defaultdict

class BM25:
    """
    A class to implement the BM25 scoring function for information retrieval.

    Attributes:
        corpus (List[str]): The corpus of documents.
        doc_freqs (Dict[str, int]): The document frequencies for each term.
        idf (Dict[str, float]): The inverse document frequencies for each term.
        doc_len (List[int]): The length of each document in the corpus.
        avg_doc_len (float): The average document length in the corpus.
        k1 (float): A parameter for tuning term frequency saturation.
        b (float): A parameter for tuning document length normalization.
    """

    def __init__(self, corpus: List[str], k1: float = 1.5, b: float = 0.75):
        """
        Initialize the BM25 class with a corpus and parameters.

        Args:
            corpus (List[str]): The corpus of documents.
            k1 (float): A parameter for tuning term frequency saturation.
            b (float): A parameter for tuning document length normalization.
        """
        self.corpus = corpus
        self.k1 = k1
        self.b = b
        self.doc_freqs = defaultdict(int)
        self.idf = {}
        self.doc_len = []
        self.avg_doc_len = 0.0
        self._initialize()

    def _initialize(self) -> None:
        """Initialize the document frequencies, IDF, and document lengths."""
        for document in self.corpus:
            doc_len = 0
            term_freqs = defaultdict(int)
            for term in document.split():
                term_freqs[term] += 1
                doc_len += 1
            self.doc_len.append(doc_len)
            for term, freq in term_freqs.items():
                self.doc_freqs[term] += 1
        self.avg_doc_len = sum(self.doc_len) / len(self.doc_len)
        for term, freq in self.doc_freqs.items():
            self.idf[term] = math.log((len(self.corpus) - freq + 0.5) / (freq + 0.5) + 1)

    def get_score(self, query: str, doc_idx: int) -> float:
        """
        Calculate the BM25 score for a query and a document.

        Args:
            query (str): The query string.
            doc_idx (int): The index of the document in the corpus.

        Returns:
            float: The BM25 score for the query and document.
        """
        score = 0.0
        doc = self.corpus[doc_idx]
        doc_terms = doc.split()
        doc_term_freqs = defaultdict(int)
        for term in doc_terms:
            doc_term_freqs[term] += 1
        for term in query.split():
            if term not in self.doc_freqs or term not in doc_term_freqs:
                continue
            tf = doc_term_freqs[term]
            idf = self.idf[term]
            numerator = tf * (self.k1 + 1)
            denominator = tf + self.k1 * (1 - self.b + self.b * self.doc_len[doc_idx] / self.avg_doc_len)
            score += idf * (numerator / denominator)
        return score

    def search(self, query: str, top_n: int = 5) -> List[Tuple[int, float]]:
        """
        Search the corpus for the top N documents matching the query.

        Args:
            query (str): The query string.
            top_n (int): The number of top documents to return.

        Returns:
            List[Tuple[int, float]]: A list of tuples containing the document index and score.
        """
        scores = [(i, self.get_score(query, i)) for i in range(len(self.corpus))]
        scores.sort(key=lambda x: x[1], reverse=True)
        return scores[:top_n]

if __name__ == "__main__":
    # Example corpus
    corpus = [
        "the quick brown fox jumps over the lazy dog",
        "the lazy dog sleeps in the sun",
        "the quick brown fox is very quick",
        "the sun is shining brightly today",
        "the dog is lazy but the fox is quick"
    ]

    # Initialize BM25
    bm25 = BM25(corpus)

    # Example query
    query = "quick fox"

    # Search the corpus
    results = bm25.search(query)

    # Print the results
    print(f"Query: {query}")
    print("Top documents:")
    for doc_idx, score in results:
        print(f"Document {doc_idx}: {corpus[doc_idx]} (Score: {score:.4f})")
