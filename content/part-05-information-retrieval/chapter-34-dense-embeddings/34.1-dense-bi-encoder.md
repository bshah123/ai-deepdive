```python
import time
from typing import List, Dict, Optional, Tuple
import torch
from sentence_transformers import SentenceTransformer, util

class BiEncoder:
    """
    A class to implement the Bi-Encoder architecture for dense embeddings using Sentence-Transformers.
    """

    def __init__(self, model_name: str = 'all-MiniLM-L6-v2'):
        """
        Initialize the Bi-Encoder with a pre-trained Sentence-Transformer model.

        Args:
            model_name (str): The name of the pre-trained model to use. Defaults to 'all-MiniLM-L6-v2'.
        """
        self.model = SentenceTransformer(model_name)

    def encode(self, sentences: List[str]) -> torch.Tensor:
        """
        Encode a list of sentences into dense embeddings.

        Args:
            sentences (List[str]): A list of sentences to encode.

        Returns:
            torch.Tensor: A tensor containing the dense embeddings of the input sentences.
        """
        return self.model.encode(sentences, convert_to_tensor=True)

    def compute_similarity(self, embeddings1: torch.Tensor, embeddings2: torch.Tensor) -> torch.Tensor:
        """
        Compute the cosine similarity between two sets of embeddings.

        Args:
            embeddings1 (torch.Tensor): The first set of embeddings.
            embeddings2 (torch.Tensor): The second set of embeddings.

        Returns:
            torch.Tensor: A tensor containing the cosine similarity scores between the input embeddings.
        """
        return util.cos_sim(embeddings1, embeddings2)

    def retrieve_top_k(self, query: str, documents: List[str], k: int = 5) -> Tuple[List[str], List[float]]:
        """
        Retrieve the top-k most similar documents to a query using the Bi-Encoder architecture.

        Args:
            query (str): The query sentence.
            documents (List[str]): A list of document sentences.
            k (int): The number of top documents to retrieve. Defaults to 5.

        Returns:
            Tuple[List[str], List[float]]: A tuple containing the top-k documents and their corresponding similarity scores.
        """
        query_embedding = self.encode([query])
        document_embeddings = self.encode(documents)

        similarity_scores = self.compute_similarity(query_embedding, document_embeddings)
        top_k_scores, top_k_indices = torch.topk(similarity_scores, k=k)

        top_k_documents = [documents[i] for i in top_k_indices[0].tolist()]
        top_k_scores = top_k_scores[0].tolist()

        return top_k_documents, top_k_scores

if __name__ == "__main__":
    # Example usage
    bi_encoder = BiEncoder()

    documents = [
        "The cat sits on the mat.",
        "The dog plays in the park.",
        "A cat is sitting on the mat.",
        "Dogs are playing in the park.",
        "The cat and the dog are friends."
    ]

    query = "A cat is on the mat."

    start_time = time.time()
    top_k_documents, top_k_scores = bi_encoder.retrieve_top_k(query, documents, k=3)
    end_time = time.time()

    print("Query:", query)
    print("Top-k Documents and Scores:")
    for doc, score in zip(top_k_documents, top_k_scores):
        print(f"Document: {doc}, Score: {score:.4f}")

    print(f"Retrieval Time: {end_time - start_time:.4f} seconds")
```
