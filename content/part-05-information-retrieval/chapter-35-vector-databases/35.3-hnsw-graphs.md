```python
import numpy as np
from typing import List, Dict, Optional, Tuple
import time
import torch

class HNSWGraph:
    """
    Hierarchical Navigable Small World (HNSW) Graph implementation for approximate nearest neighbor search.
    """

    def __init__(self, dim: int, m: int = 16, ef_construction: int = 200, max_layers: int = 16):
        """
        Initialize the HNSW graph.

        Args:
            dim (int): Dimensionality of the vectors.
            m (int): Number of bi-directional links created for every new element during construction.
            ef_construction (int): Size of the dynamic candidate list for construction.
            max_layers (int): Maximum number of layers in the graph.
        """
        self.dim = dim
        self.m = m
        self.ef_construction = ef_construction
        self.max_layers = max_layers
        self.graph: Dict[int, Dict[int, List[int]]] = {0: {}}  # Layer 0 is the base layer
        self.enter_point: Optional[int] = None
        self.vectors: Dict[int, np.ndarray] = {}
        self.id_counter = 0

    def _distance(self, a: np.ndarray, b: np.ndarray) -> float:
        """
        Compute the Euclidean distance between two vectors.

        Args:
            a (np.ndarray): First vector.
            b (np.ndarray): Second vector.

        Returns:
            float: Euclidean distance between the vectors.
        """
        return np.linalg.norm(a - b)

    def _select_neighbors(self, q: np.ndarray, candidates: List[int], k: int) -> List[int]:
        """
        Select the k nearest neighbors from the candidates.

        Args:
            q (np.ndarray): Query vector.
            candidates (List[int]): List of candidate indices.
            k (int): Number of neighbors to select.

        Returns:
            List[int]: Indices of the k nearest neighbors.
        """
        distances = [(idx, self._distance(q, self.vectors[idx])) for idx in candidates]
        distances.sort(key=lambda x: x[1])
        return [idx for idx, _ in distances[:k]]

    def _search_layer(self, q: np.ndarray, entry_point: int, ef: int, layer: int) -> List[int]:
        """
        Search for the nearest neighbors in a specific layer.

        Args:
            q (np.ndarray): Query vector.
            entry_point (int): Index of the entry point in the layer.
            ef (int): Size of the dynamic candidate list.
            layer (int): Layer to search in.

        Returns:
            List[int]: Indices of the nearest neighbors in the layer.
        """
        visited = set()
        candidates = []
        heap = [(self._distance(q, self.vectors[entry_point]), entry_point)]
        visited.add(entry_point)

        while heap:
            dist, idx = heap.pop(0)
            if len(candidates) >= ef and dist > self._distance(q, self.vectors[candidates[-1]]):
                break
            candidates.append(idx)
            for neighbor in self.graph[layer][idx]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    heap.append((self._distance(q, self.vectors[neighbor]), neighbor))
                    heap.sort(key=lambda x: x[0])

        return self._select_neighbors(q, candidates, ef)

    def add_vector(self, vector: np.ndarray) -> int:
        """
        Add a vector to the HNSW graph.

        Args:
            vector (np.ndarray): Vector to add.

        Returns:
            int: Index of the added vector.
        """
        if len(vector) != self.dim:
            raise ValueError(f"Vector dimension must be {self.dim}")

        idx = self.id_counter
        self.id_counter += 1
        self.vectors[idx] = vector

        if self.enter_point is None:
            self.enter_point = idx
            for layer in range(1, self.max_layers):
                self.graph[layer] = {idx: []}
            return idx

        # Determine the layer for the new vector
        layer = min(int(np.log2(idx + 1)), self.max_layers - 1)

        # Add the vector to the graph
        if layer not in self.graph:
            self.graph[layer] = {}

        self.graph[layer][idx] = []

        # Search for the nearest neighbors in the upper layers
        entry_point = self.enter_point
        for l in range(self.max_layers - 1, layer, -1):
            if l in self.graph and entry_point in self.graph[l]:
                neighbors = self._search_layer(vector, entry_point, 1, l)
                if neighbors:
                    entry_point = neighbors[0]

        # Search for the nearest neighbors in the current layer
        candidates = self._search_layer(vector, entry_point, self.ef_construction, layer)
        neighbors = self._select_neighbors(vector, candidates, self.m)

        # Add bi-directional links to the neighbors
        for neighbor in neighbors:
            self.graph[layer][idx].append(neighbor)
            self.graph[layer][neighbor].append(idx)

        # Update the enter point if necessary
        if layer == self.max_layers - 1:
            self.enter_point = idx

        return idx

    def search(self, q: np.ndarray, k: int = 1, ef: int = 200) -> List[int]:
        """
        Search for the k nearest neighbors of the query vector.

        Args:
            q (np.ndarray): Query vector.
            k (int): Number of neighbors to return.
            ef (int): Size of the dynamic candidate list.

        Returns:
            List[int]: Indices of the k nearest neighbors.
        """
        if len(q) != self.dim:
            raise ValueError(f"Query vector dimension must be {self.dim}")

        if self.enter_point is None:
            return []

        # Search for the nearest neighbors in the upper layers
        entry_point = self.enter_point
        for layer in range(self.max_layers - 1, -1, -1):
            if layer in self.graph and entry_point in self.graph[layer]:
                neighbors = self._search_layer(q, entry_point, 1, layer)
                if neighbors:
                    entry_point = neighbors[0]

        # Search for the nearest neighbors in the base layer
        candidates = self._search_layer(q, entry_point, ef, 0)
        return self._select_neighbors(q, candidates, k)

if __name__ == "__main__":
    # Demonstration code
    dim = 128
    num_vectors = 1000
    k = 5
    ef = 200

    # Generate random vectors
    vectors = torch.randn(num_vectors, dim).numpy()

    # Create HNSW graph
    hnsw = HNSWGraph(dim=dim, ef_construction=ef)

    # Add vectors to the graph
    start_time = time.time()
    for vector in vectors:
        hnsw.add_vector(vector)
    add_time = time.time() - start_time

    # Search for the nearest neighbors
    query = torch.randn(1, dim).numpy()[0]
    start_time = time.time()
    neighbors = hnsw.search(query, k=k, ef=ef)
    search_time = time.time() - start_time

    # Print results
    print(f"Added {num_vectors} vectors to the HNSW graph in {add_time:.4f} seconds")
    print(f"Searched for {k} nearest neighbors in {search_time:.4f} seconds")
    print(f"Nearest neighbors: {neighbors}")
```
