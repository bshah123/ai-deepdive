```python
import time
from typing import List, Dict, Optional, Tuple
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

class MinimalistRAGSystem:
    """
    A minimalist Retrieval-Augmented Generation (RAG) system implemented in pure Python.
    This system uses TF-IDF for document retrieval and a simple template-based generation.
    """

    def __init__(self, documents: List[str]):
        """
        Initialize the RAG system with a list of documents.

        Args:
            documents: A list of strings representing the documents to be indexed.
        """
        self.documents = documents
        self.vectorizer = TfidfVectorizer()
        self.doc_vectors = self.vectorizer.fit_transform(documents)

    def retrieve(self, query: str, top_k: int = 3) -> List[Tuple[int, float]]:
        """
        Retrieve the top-k most relevant documents for a given query.

        Args:
            query: The query string.
            top_k: The number of top documents to retrieve.

        Returns:
            A list of tuples containing the document index and its relevance score.
        """
        query_vector = self.vectorizer.transform([query])
        similarities = cosine_similarity(query_vector, self.doc_vectors).flatten()
        top_indices = np.argsort(similarities)[-top_k:][::-1]
        return [(idx, similarities[idx]) for idx in top_indices]

    def generate(self, query: str, retrieved_docs: List[Tuple[int, float]]) -> str:
        """
        Generate a response based on the retrieved documents and the query.

        Args:
            query: The query string.
            retrieved_docs: A list of tuples containing the document index and its relevance score.

        Returns:
            A generated response string.
        """
        context = "\n".join([self.documents[idx] for idx, _ in retrieved_docs])
        return f"Based on the provided documents, here is the answer to your query '{query}':\n\n{context}"

    def rag_pipeline(self, query: str, top_k: int = 3) -> str:
        """
        Execute the complete RAG pipeline: retrieve and generate.

        Args:
            query: The query string.
            top_k: The number of top documents to retrieve.

        Returns:
            A generated response string.
        """
        retrieved_docs = self.retrieve(query, top_k)
        return self.generate(query, retrieved_docs)

if __name__ == "__main__":
    # Example documents
    documents = [
        "The capital of France is Paris.",
        "Paris is known for the Eiffel Tower.",
        "The Eiffel Tower was completed in 1889.",
        "The capital of Germany is Berlin.",
        "Berlin is known for the Brandenburg Gate.",
        "The Brandenburg Gate was built in the 18th century."
    ]

    # Initialize the RAG system
    rag_system = MinimalistRAGSystem(documents)

    # Example queries
    queries = [
        "What is the capital of France?",
        "Tell me about the Eiffel Tower.",
        "When was the Eiffel Tower completed?"
    ]

    # Execute the RAG pipeline for each query
    for query in queries:
        start_time = time.time()
        response = rag_system.rag_pipeline(query)
        end_time = time.time()
        print(f"Query: {query}")
        print(f"Response: {response}")
        print(f"Time taken: {end_time - start_time:.4f} seconds\n")
```
