```python
import os
import time
from typing import List, Dict, Optional, Tuple
from pathlib import Path
import PyPDF2
import pandas as pd
import markdown
from bs4 import BeautifulSoup
from pdfminer.high_level import extract_pages
from pdfminer.layout import LTTextContainer, LTFigure, LTImage

class DocumentParser:
    """
    A class to parse various document formats including PDFs, tables, and Markdown.
    Preserves layout information where possible.
    """

    def __init__(self):
        self.parsed_data = []

    def parse_pdf(self, file_path: str) -> List[Dict[str, str]]:
        """
        Parse a PDF file and extract text and layout information.

        Args:
            file_path (str): Path to the PDF file.

        Returns:
            List[Dict[str, str]]: List of dictionaries containing text and layout information.
        """
        parsed_data = []
        with open(file_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            for page_num, page in enumerate(reader.pages):
                text = page.extract_text()
                parsed_data.append({
                    'page': page_num + 1,
                    'text': text,
                    'layout': 'text'
                })

        # Extract layout information using pdfminer
        for page_layout in extract_pages(file_path):
            for element in page_layout:
                if isinstance(element, LTTextContainer):
                    for text_line in element:
                        parsed_data.append({
                            'page': page_layout.pageid,
                            'text': text_line.get_text(),
                            'layout': 'text'
                        })
                elif isinstance(element, LTFigure):
                    parsed_data.append({
                        'page': page_layout.pageid,
                        'text': 'Figure',
                        'layout': 'figure'
                    })
                elif isinstance(element, LTImage):
                    parsed_data.append({
                        'page': page_layout.pageid,
                        'text': 'Image',
                        'layout': 'image'
                    })

        self.parsed_data.extend(parsed_data)
        return parsed_data

    def parse_table(self, file_path: str) -> List[Dict[str, str]]:
        """
        Parse a table from a CSV or Excel file.

        Args:
            file_path (str): Path to the table file.

        Returns:
            List[Dict[str, str]]: List of dictionaries containing table data.
        """
        if file_path.endswith('.csv'):
            df = pd.read_csv(file_path)
        elif file_path.endswith(('.xlsx', '.xls')):
            df = pd.read_excel(file_path)
        else:
            raise ValueError("Unsupported file format for table parsing.")

        parsed_data = df.to_dict('records')
        self.parsed_data.extend(parsed_data)
        return parsed_data

    def parse_markdown(self, file_path: str) -> List[Dict[str, str]]:
        """
        Parse a Markdown file and extract text and layout information.

        Args:
            file_path (str): Path to the Markdown file.

        Returns:
            List[Dict[str, str]]: List of dictionaries containing text and layout information.
        """
        with open(file_path, 'r', encoding='utf-8') as file:
            md_content = file.read()

        html_content = markdown.markdown(md_content)
        soup = BeautifulSoup(html_content, 'html.parser')

        parsed_data = []
        for element in soup.find_all(['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'p', 'ul', 'ol', 'table']):
            parsed_data.append({
                'text': element.get_text(),
                'layout': element.name
            })

        self.parsed_data.extend(parsed_data)
        return parsed_data

    def get_parsed_data(self) -> List[Dict[str, str]]:
        """
        Get the parsed data from all files.

        Returns:
            List[Dict[str, str]]: List of dictionaries containing parsed data.
        """
        return self.parsed_data

if __name__ == "__main__":
    # Example usage
    parser = DocumentParser()

    # Parse PDF
    pdf_path = 'example.pdf'
    if os.path.exists(pdf_path):
        start_time = time.time()
        pdf_data = parser.parse_pdf(pdf_path)
        print(f"PDF Parsing Time: {time.time() - start_time:.2f} seconds")
        print(f"PDF Data: {pdf_data[:2]}")  # Print first 2 entries for brevity

    # Parse Table
    table_path = 'example.csv'
    if os.path.exists(table_path):
        start_time = time.time()
        table_data = parser.parse_table(table_path)
        print(f"Table Parsing Time: {time.time() - start_time:.2f} seconds")
        print(f"Table Data: {table_data[:2]}")  # Print first 2 entries for brevity

    # Parse Markdown
    md_path = 'example.md'
    if os.path.exists(md_path):
        start_time = time.time()
        md_data = parser.parse_markdown(md_path)
        print(f"Markdown Parsing Time: {time.time() - start_time:.2f} seconds")
        print(f"Markdown Data: {md_data[:2]}")  # Print first 2 entries for brevity

    # Get all parsed data
    all_data = parser.get_parsed_data()
    print(f"Total Parsed Data Entries: {len(all_data)}")
```
