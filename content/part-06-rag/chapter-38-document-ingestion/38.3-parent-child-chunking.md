```python
import os
import time
from typing import List, Dict, Optional, Tuple
from pathlib import Path
from datetime import datetime
from PyPDF2 import PdfReader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.schema import Document

class DocumentProcessor:
    """
    A class to process documents, extract metadata, and perform hierarchical chunking.
    """

    def __init__(self, chunk_size: int = 1000, chunk_overlap: int = 200):
        """
        Initialize the DocumentProcessor with chunking parameters.

        Args:
            chunk_size (int): The size of each chunk.
            chunk_overlap (int): The overlap between consecutive chunks.
        """
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=self.chunk_size,
            chunk_overlap=self.chunk_overlap
        )

    def extract_metadata(self, file_path: str) -> Dict:
        """
        Extract metadata from a PDF file.

        Args:
            file_path (str): The path to the PDF file.

        Returns:
            Dict: A dictionary containing the extracted metadata.
        """
        metadata = {}
        with open(file_path, 'rb') as file:
            reader = PdfReader(file)
            metadata = reader.metadata
            metadata['file_path'] = file_path
            metadata['file_name'] = os.path.basename(file_path)
            metadata['file_size'] = os.path.getsize(file_path)
            metadata['creation_date'] = metadata.get('/CreationDate', '')
            metadata['modification_date'] = metadata.get('/ModDate', '')
            metadata['author'] = metadata.get('/Author', '')
            metadata['title'] = metadata.get('/Title', '')
            metadata['subject'] = metadata.get('/Subject', '')
            metadata['keywords'] = metadata.get('/Keywords', '')
        return metadata

    def hierarchical_chunking(self, text: str, metadata: Dict) -> List[Document]:
        """
        Perform hierarchical chunking on the text.

        Args:
            text (str): The text to be chunked.
            metadata (Dict): The metadata associated with the text.

        Returns:
            List[Document]: A list of Document objects with hierarchical chunks.
        """
        chunks = self.text_splitter.split_text(text)
        documents = []
        for i, chunk in enumerate(chunks):
            doc_metadata = metadata.copy()
            doc_metadata['chunk_id'] = i
            doc_metadata['chunk_size'] = len(chunk)
            doc_metadata['chunk_parent'] = None if i == 0 else i - 1
            documents.append(Document(page_content=chunk, metadata=doc_metadata))
        return documents

    def process_document(self, file_path: str) -> List[Document]:
        """
        Process a document by extracting metadata and performing hierarchical chunking.

        Args:
            file_path (str): The path to the document file.

        Returns:
            List[Document]: A list of Document objects with hierarchical chunks.
        """
        metadata = self.extract_metadata(file_path)
        text = ""
        with open(file_path, 'rb') as file:
            reader = PdfReader(file)
            for page in reader.pages:
                text += page.extract_text()
        documents = self.hierarchical_chunking(text, metadata)
        return documents

if __name__ == "__main__":
    # Example usage
    file_path = "example.pdf"
    processor = DocumentProcessor(chunk_size=1000, chunk_overlap=200)

    start_time = time.time()
    documents = processor.process_document(file_path)
    end_time = time.time()

    print(f"Processed document: {file_path}")
    print(f"Total chunks created: {len(documents)}")
    print(f"Processing time: {end_time - start_time:.2f} seconds")

    # Print metadata and content of the first chunk
    if documents:
        first_chunk = documents[0]
        print("\nMetadata of the first chunk:")
        for key, value in first_chunk.metadata.items():
            print(f"{key}: {value}")
        print("\nContent of the first chunk:")
        print(first_chunk.page_content[:500] + "...")  # Print first 500 characters
```
