from typing import List, Dict, Optional, Tuple
import torch
from torch import nn, Tensor
from transformers import AutoTokenizer, AutoModel
import time
import numpy as np

class BiEncoder(nn.Module):
    """
    A bi-encoder model for encoding queries and documents separately.
    """
    def __init__(self, model_name: str = 'sentence-transformers/all-MiniLM-L6-v2'):
        super().__init__()
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name)

    def forward(self, input_ids: Tensor, attention_mask: Tensor) -> Tensor:
        """
        Forward pass of the bi-encoder model.

        Args:
            input_ids: Tensor of token ids.
            attention_mask: Tensor of attention masks.

        Returns:
            Tensor of embeddings.
        """
        outputs = self.model(input_ids=input_ids, attention_mask=attention_mask)
        return outputs.last_hidden_state[:, 0, :]

    def encode(self, texts: List[str]) -> Tensor:
        """
        Encode a list of texts into embeddings.

        Args:
            texts: List of texts to encode.

        Returns:
            Tensor of embeddings.
        """
        inputs = self.tokenizer(texts, padding=True, truncation=True, return_tensors='pt')
        with torch.no_grad():
            embeddings = self.forward(inputs['input_ids'], inputs['attention_mask'])
        return embeddings

class CrossEncoder(nn.Module):
    """
    A cross-encoder model for scoring query-document pairs.
    """
    def __init__(self, model_name: str = 'cross-encoder/ms-marco-MiniLM-L-6-v2'):
        super().__init__()
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name)

    def forward(self, input_ids: Tensor, attention_mask: Tensor) -> Tensor:
        """
        Forward pass of the cross-encoder model.

        Args:
            input_ids: Tensor of token ids.
            attention_mask: Tensor of attention masks.

        Returns:
            Tensor of scores.
        """
        outputs = self.model(input_ids=input_ids, attention_mask=attention_mask)
        return outputs.logits

    def score(self, queries: List[str], documents: List[str]) -> Tensor:
        """
        Score a list of query-document pairs.

        Args:
            queries: List of queries.
            documents: List of documents.

        Returns:
            Tensor of scores.
        """
        inputs = self.tokenizer(queries, documents, padding=True, truncation=True, return_tensors='pt')
        with torch.no_grad():
            scores = self.forward(inputs['input_ids'], inputs['attention_mask'])
        return scores

def re_rank(documents: List[str], query: str, bi_encoder: BiEncoder, cross_encoder: CrossEncoder, top_k: int = 5) -> List[Tuple[str, float]]:
    """
    Re-rank a list of documents based on a query using a bi-encoder and a cross-encoder.

    Args:
        documents: List of documents to re-rank.
        query: Query to use for re-ranking.
        bi_encoder: Bi-encoder model.
        cross_encoder: Cross-encoder model.
        top_k: Number of top documents to return.

    Returns:
        List of tuples containing the top-k documents and their scores.
    """
    # Encode the query and documents using the bi-encoder
    query_embedding = bi_encoder.encode([query])
    document_embeddings = bi_encoder.encode(documents)

    # Compute the cosine similarity between the query and each document
    similarities = torch.nn.functional.cosine_similarity(query_embedding, document_embeddings)

    # Get the top-k documents based on the cosine similarity
    top_k_indices = torch.topk(similarities, top_k).indices
    top_k_documents = [documents[i] for i in top_k_indices]

    # Score the top-k documents using the cross-encoder
    scores = cross_encoder.score([query] * len(top_k_documents), top_k_documents)

    # Combine the documents and their scores
    ranked_documents = list(zip(top_k_documents, scores.squeeze().tolist()))

    # Sort the documents by their scores in descending order
    ranked_documents.sort(key=lambda x: x[1], reverse=True)

    return ranked_documents

if __name__ == "__main__":
    # Initialize the bi-encoder and cross-encoder models
    bi_encoder = BiEncoder()
    cross_encoder = CrossEncoder()

    # Example documents and query
    documents = [
        "The quick brown fox jumps over the lazy dog.",
        "A quick brown dog outpaces a fast fox.",
        "The lazy dog sleeps all day.",
        "The fox is quick and the dog is lazy.",
        "A fast fox jumps over a sleeping dog."
    ]
    query = "What is the relationship between the fox and the dog?"

    # Re-rank the documents based on the query
    start_time = time.time()
    ranked_documents = re_rank(documents, query, bi_encoder, cross_encoder)
    end_time = time.time()

    # Print the ranked documents and their scores
    print("Ranked Documents:")
    for i, (document, score) in enumerate(ranked_documents, 1):
        print(f"{i}. Score: {score:.4f} - Document: {document}")

    # Print the time taken to re-rank the documents
    print(f"\nTime taken to re-rank documents: {end_time - start_time:.4f} seconds")

    # Print the memory footprint of the bi-encoder and cross-encoder models
    print("\nMemory Footprint:")
    print(f"Bi-encoder: {sum(p.numel() for p in bi_encoder.parameters()) / 1e6:.2f} million parameters")
    print(f"Cross-encoder: {sum(p.numel() for p in cross_encoder.parameters()) / 1e6:.2f} million parameters")
