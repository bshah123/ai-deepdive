import time
from typing import List, Dict, Optional, Tuple
from functools import lru_cache
import numpy as np
import torch

class ContextCache:
    """
    A class to manage context caching for RAG applications, optimizing cost and performance.
    """

    def __init__(self, max_cache_size: int = 1000):
        """
        Initialize the ContextCache with a maximum cache size.

        Args:
            max_cache_size (int): Maximum number of items to keep in the cache.
        """
        self.max_cache_size = max_cache_size
        self.cache: Dict[str, Tuple[List[str], float]] = {}
        self.cache_order: List[str] = []

    @lru_cache(maxsize=None)
    def _compute_similarity(self, context1: List[str], context2: List[str]) -> float:
        """
        Compute the similarity between two contexts using cosine similarity.

        Args:
            context1 (List[str]): First context.
            context2 (List[str]): Second context.

        Returns:
            float: Cosine similarity between the two contexts.
        """
        # Convert contexts to vectors
        vector1 = self._context_to_vector(context1)
        vector2 = self._context_to_vector(context2)

        # Compute cosine similarity
        dot_product = np.dot(vector1, vector2)
        norm1 = np.linalg.norm(vector1)
        norm2 = np.linalg.norm(vector2)

        if norm1 == 0 or norm2 == 0:
            return 0.0

        return dot_product / (norm1 * norm2)

    def _context_to_vector(self, context: List[str]) -> np.ndarray:
        """
        Convert a context to a vector representation.

        Args:
            context (List[str]): Context to convert.

        Returns:
            np.ndarray: Vector representation of the context.
        """
        # Simple implementation: average of word embeddings
        word_embeddings = [self._get_word_embedding(word) for word in context]
        return np.mean(word_embeddings, axis=0)

    def _get_word_embedding(self, word: str) -> np.ndarray:
        """
        Get the embedding for a word. This is a placeholder for a real embedding model.

        Args:
            word (str): Word to get embedding for.

        Returns:
            np.ndarray: Embedding for the word.
        """
        # Placeholder: random embedding for demonstration
        return np.random.rand(300)

    def add_to_cache(self, query: str, context: List[str]) -> None:
        """
        Add a query and its context to the cache.

        Args:
            query (str): Query to cache.
            context (List[str]): Context to cache.
        """
        if query in self.cache:
            self.cache_order.remove(query)

        self.cache[query] = (context, time.time())
        self.cache_order.insert(0, query)

        if len(self.cache) > self.max_cache_size:
            oldest_query = self.cache_order.pop()
            del self.cache[oldest_query]

    def get_cached_context(self, query: str, similarity_threshold: float = 0.7) -> Optional[List[str]]:
        """
        Retrieve a cached context for a given query if it exists and is similar enough.

        Args:
            query (str): Query to retrieve context for.
            similarity_threshold (float): Minimum similarity threshold for cached context.

        Returns:
            Optional[List[str]]: Cached context if found and similar enough, None otherwise.
        """
        if query in self.cache:
            cached_context, _ = self.cache[query]
            similarity = self._compute_similarity(query.split(), cached_context)
            if similarity >= similarity_threshold:
                return cached_context

        for cached_query, (cached_context, _) in self.cache.items():
            similarity = self._compute_similarity(query.split(), cached_query.split())
            if similarity >= similarity_threshold:
                return cached_context

        return None

    def clear_cache(self) -> None:
        """
        Clear the cache.
        """
        self.cache.clear()
        self.cache_order.clear()

def demonstrate_context_caching():
    """
    Demonstrate the ContextCache class with realistic test inputs.
    """
    cache = ContextCache(max_cache_size=5)

    # Test inputs
    queries = [
        "What is the capital of France?",
        "Tell me about the Eiffel Tower.",
        "Explain the concept of machine learning.",
        "What are the benefits of using Python?",
        "How does quantum computing work?",
        "What is the capital of France again?",
        "Can you explain machine learning in simple terms?"
    ]

    contexts = [
        ["The capital of France is Paris."],
        ["The Eiffel Tower is a wrought-iron lattice tower on the Champ de Mars in Paris, France."],
        ["Machine learning is a subset of artificial intelligence that provides systems the ability to automatically learn and improve from experience without being explicitly programmed."],
        ["Python is a high-level, interpreted, general-purpose programming language."],
        ["Quantum computing uses quantum-mechanical phenomena such as superposition and entanglement to perform computation."],
        ["The capital of France is Paris."],
        ["Machine learning is a method of data analysis that automates analytical model building."]
    ]

    # Add queries and contexts to cache
    for query, context in zip(queries, contexts):
        cache.add_to_cache(query, context)

    # Retrieve cached contexts
    for query in queries:
        start_time = time.time()
        cached_context = cache.get_cached_context(query)
        end_time = time.time()

        print(f"Query: {query}")
        print(f"Cached Context: {cached_context}")
        print(f"Time taken: {end_time - start_time:.6f} seconds\n")

if __name__ == "__main__":
    demonstrate_context_caching()
