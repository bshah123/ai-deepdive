```python
from typing import List, Dict, Optional, Tuple
import time
import semantic_kernel as sk
from semantic_kernel.connectors.ai.open_ai import OpenAIChatCompletion

class NativeFunctionPlugin:
    """
    A class to represent a native function plugin for Semantic Kernel.
    This plugin contains native functions that can be used within the kernel.
    """

    def __init__(self, kernel: sk.Kernel):
        """
        Initialize the NativeFunctionPlugin with a Semantic Kernel.

        Args:
            kernel (sk.Kernel): The Semantic Kernel instance.
        """
        self.kernel = kernel

    def greet(self, name: str) -> str:
        """
        A native function that greets the user.

        Args:
            name (str): The name of the user to greet.

        Returns:
            str: A greeting message.
        """
        return f"Hello, {name}! Welcome to Semantic Kernel."

    def add_numbers(self, a: float, b: float) -> float:
        """
        A native function that adds two numbers.

        Args:
            a (float): The first number.
            b (float): The second number.

        Returns:
            float: The sum of the two numbers.
        """
        return a + b

    def multiply_numbers(self, a: float, b: float) -> float:
        """
        A native function that multiplies two numbers.

        Args:
            a (float): The first number.
            b (float): The second number.

        Returns:
            float: The product of the two numbers.
        """
        return a * b

class SemanticKernelOrchestrator:
    """
    A class to orchestrate the Semantic Kernel and its plugins.
    """

    def __init__(self, api_key: str, org_id: str):
        """
        Initialize the SemanticKernelOrchestrator with API key and organization ID.

        Args:
            api_key (str): The OpenAI API key.
            org_id (str): The OpenAI organization ID.
        """
        self.kernel = sk.Kernel()
        self.kernel.add_chat_service(
            "chat-gpt",
            OpenAIChatCompletion("gpt-3.5-turbo", api_key, org_id)
        )
        self.plugin = NativeFunctionPlugin(self.kernel)

    async def run(self) -> None:
        """
        Run the Semantic Kernel with the native function plugin.
        """
        print("Running Semantic Kernel with Native Function Plugin...")

        # Demonstrate native function calls
        print("\nDemonstrating native function calls:")
        print(f"Greeting: {self.plugin.greet('Alice')}")
        print(f"Addition: 5 + 3 = {self.plugin.add_numbers(5, 3)}")
        print(f"Multiplication: 5 * 3 = {self.plugin.multiply_numbers(5, 3)}")

        # Demonstrate semantic function calls
        print("\nDemonstrating semantic function calls:")
        prompt = """
        {{$input}}

        Summarize the above text.
        """
        summarize_function = self.kernel.create_semantic_function(prompt)
        summary = await summarize_function("The quick brown fox jumps over the lazy dog.")
        print(f"Summary: {summary}")

if __name__ == "__main__":
    import asyncio

    # Replace with your actual OpenAI API key and organization ID
    API_KEY = "your-openai-api-key"
    ORG_ID = "your-openai-org-id"

    orchestrator = SemanticKernelOrchestrator(API_KEY, ORG_ID)

    start_time = time.time()
    asyncio.run(orchestrator.run())
    end_time = time.time()

    print(f"\nExecution time: {end_time - start_time:.2f} seconds")
```
