import numpy as np
from typing import List, Dict, Tuple, Optional
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import time

class RAGTriadEvaluator:
    """
    A class to evaluate the RAG Triad: Context Relevance, Groundedness, and Answer Relevance.
    """

    def __init__(self):
        """
        Initialize the RAGTriadEvaluator with a TF-IDF vectorizer.
        """
        self.vectorizer = TfidfVectorizer()

    def context_relevance(self, question: str, contexts: List[str]) -> List[float]:
        """
        Evaluate the relevance of each context to the question.

        Args:
            question (str): The question to evaluate.
            contexts (List[str]): List of contexts to evaluate.

        Returns:
            List[float]: List of relevance scores for each context.
        """
        if not contexts:
            return []

        # Combine question and contexts for vectorization
        texts = [question] + contexts
        tfidf_matrix = self.vectorizer.fit_transform(texts)

        # Calculate cosine similarity between question and each context
        question_vector = tfidf_matrix[0:1]
        context_vectors = tfidf_matrix[1:]
        similarities = cosine_similarity(question_vector, context_vectors).flatten()

        return similarities.tolist()

    def groundedness(self, answer: str, contexts: List[str]) -> float:
        """
        Evaluate the groundedness of the answer in the contexts.

        Args:
            answer (str): The answer to evaluate.
            contexts (List[str]): List of contexts to evaluate.

        Returns:
            float: Groundedness score.
        """
        if not contexts:
            return 0.0

        # Combine answer and contexts for vectorization
        texts = [answer] + contexts
        tfidf_matrix = self.vectorizer.fit_transform(texts)

        # Calculate cosine similarity between answer and each context
        answer_vector = tfidf_matrix[0:1]
        context_vectors = tfidf_matrix[1:]
        similarities = cosine_similarity(answer_vector, context_vectors).flatten()

        return np.max(similarities)

    def answer_relevance(self, question: str, answer: str) -> float:
        """
        Evaluate the relevance of the answer to the question.

        Args:
            question (str): The question to evaluate.
            answer (str): The answer to evaluate.

        Returns:
            float: Relevance score.
        """
        # Combine question and answer for vectorization
        texts = [question, answer]
        tfidf_matrix = self.vectorizer.fit_transform(texts)

        # Calculate cosine similarity between question and answer
        question_vector = tfidf_matrix[0:1]
        answer_vector = tfidf_matrix[1:]
        similarity = cosine_similarity(question_vector, answer_vector).flatten()[0]

        return similarity

    def evaluate_rag_triad(self, question: str, contexts: List[str], answer: str) -> Dict[str, float]:
        """
        Evaluate the RAG Triad: Context Relevance, Groundedness, and Answer Relevance.

        Args:
            question (str): The question to evaluate.
            contexts (List[str]): List of contexts to evaluate.
            answer (str): The answer to evaluate.

        Returns:
            Dict[str, float]: Dictionary containing the evaluation scores.
        """
        start_time = time.time()

        context_relevance_scores = self.context_relevance(question, contexts)
        groundedness_score = self.groundedness(answer, contexts)
        answer_relevance_score = self.answer_relevance(question, answer)

        end_time = time.time()
        evaluation_time = end_time - start_time

        return {
            "context_relevance_scores": context_relevance_scores,
            "groundedness_score": groundedness_score,
            "answer_relevance_score": answer_relevance_score,
            "evaluation_time": evaluation_time
        }

if __name__ == "__main__":
    # Example usage
    evaluator = RAGTriadEvaluator()

    question = "What is the capital of France?"
    contexts = [
        "France is a country in Europe.",
        "The capital of France is Paris.",
        "Paris is known for its art and culture."
    ]
    answer = "The capital of France is Paris."

    evaluation_results = evaluator.evaluate_rag_triad(question, contexts, answer)

    print("Evaluation Results:")
    print(f"Context Relevance Scores: {evaluation_results['context_relevance_scores']}")
    print(f"Groundedness Score: {evaluation_results['groundedness_score']}")
    print(f"Answer Relevance Score: {evaluation_results['answer_relevance_score']}")
    print(f"Evaluation Time: {evaluation_results['evaluation_time']:.4f} seconds")
